! ==========================================
! Router R1 Configuration
! Lab: Configuring Cisco Router Interfaces (PTL-0002)
! ==========================================
enable
configure terminal
hostname R1

! Enable IPv6 unicast packet routing globally
ipv6 unicast-routing

! LAN 1 Interface (Admin LAN)
interface GigabitEthernet0/0/0
 description Admin LAN Connection to S1
 ip address 192.168.1.1 255.255.255.0
 ipv6 address 2001:db8:acad:1::1/64
 ipv6 address fe80::1 link-local
 no shutdown
 exit

! LAN 2 Interface (Sales LAN)
interface GigabitEthernet0/0/1
 description Sales LAN Connection to S2
 ip address 192.168.2.1 255.255.255.0
 ipv6 address 2001:db8:acad:2::1/64
 ipv6 address fe80::1 link-local
 no shutdown
 exit

end
copy running-config startup-config
