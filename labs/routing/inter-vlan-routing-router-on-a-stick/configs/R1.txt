! ==========================================
! Router R1 Configuration
! Lab: Router-on-a-Stick Inter-VLAN Routing (PTL-0010)
! ==========================================
enable
configure terminal
hostname R1

! Physical Parent Interface (no IP address)
interface GigabitEthernet0/0/0
 description ROAS Physical Trunk to Switch S1
 no ip address
 no shutdown
 exit

! Subinterface for VLAN 10 (Faculty)
interface GigabitEthernet0/0/0.10
 description Default Gateway for Faculty VLAN 10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 exit

! Subinterface for VLAN 20 (Students)
interface GigabitEthernet0/0/0.20
 description Default Gateway for Student VLAN 20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 exit

end
copy running-config startup-config
