! ==========================================
! Switch S1 (Enterprise Access Switch)
! Lab: Static NAT and Inside Server Publishing (PTL-0021)
! ==========================================
enable
configure terminal
hostname S1
no ip domain-lookup

! Management SVI
interface Vlan1
 description Management SVI
 ip address 192.168.10.2 255.255.255.0
 no shutdown
 exit

! Default Gateway pointing to Edge Router R1
ip default-gateway 192.168.10.1

! Host Access Ports
interface FastEthernet0/1
 description Access Port to PC-Internal
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

interface FastEthernet0/10
 description Access Port to Svr-Internal (Web/File Server)
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

interface GigabitEthernet0/1
 description Uplink to R1 G0/0/0
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

end
copy running-config startup-config
