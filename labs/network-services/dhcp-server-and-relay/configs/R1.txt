! ==========================================
! Router R1 (Branch Gateway & DHCP Relay Agent)
! Lab: Configuring Cisco IOS DHCPv4 Server and Relay Agent (PTL-0012)
! ==========================================
enable
configure terminal
hostname R1

! LAN 1 Interface (Client Subnet with DHCP Relay)
interface GigabitEthernet0/0/0
 description Branch Client LAN
 ip address 192.168.10.1 255.255.255.0
 ! Forward client DHCP broadcasts across routed link to Central DHCP Server R2
 ip helper-address 10.0.0.2
 no shutdown
 exit

! Routed WAN Transit Interface to Central Site R2
interface GigabitEthernet0/0/1
 description Routed Transit Link to R2
 ip address 10.0.0.1 255.255.255.252
 no shutdown
 exit

! Static Route to Central Server LAN
ip route 192.168.20.0 255.255.255.0 10.0.0.2

end
copy running-config startup-config
