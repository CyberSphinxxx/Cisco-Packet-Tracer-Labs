! ==========================================
! Switch S2 (Datacenter Server Switch)
! Lab: DNS Records and HTTP Web Services (PTL-0022)
! ==========================================
enable
configure terminal
hostname S2
no ip domain-lookup

! Management SVI
interface Vlan1
 description Server Switch Management
 ip address 192.168.20.2 255.255.255.0
 no shutdown
 exit

! Default Gateway pointing to R1
ip default-gateway 192.168.20.1

! Server Access Ports
interface FastEthernet0/10
 description Access Port to DNS-Server
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

interface FastEthernet0/11
 description Access Port to Web-Server
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

interface GigabitEthernet0/1
 description Uplink to R1 G0/0/1
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

end
copy running-config startup-config
