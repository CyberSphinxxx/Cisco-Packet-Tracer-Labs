! ==========================================
! Switch S1 (Client Access Switch)
! Lab: DNS Records and HTTP Web Services (PTL-0022)
! ==========================================
enable
configure terminal
hostname S1
no ip domain-lookup

! Management SVI
interface Vlan1
 description Client Switch Management
 ip address 192.168.10.2 255.255.255.0
 no shutdown
 exit

! Default Gateway pointing to R1
ip default-gateway 192.168.10.1

! Access Ports for Clients
interface FastEthernet0/1
 description Access Port to PC-1
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

interface FastEthernet0/2
 description Access Port to PC-2
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

interface GigabitEthernet0/1
 description Uplink to R1 G0/0/0
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

end
copy running-config startup-config
